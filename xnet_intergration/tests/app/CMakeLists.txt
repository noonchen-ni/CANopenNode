cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(xnet_canopen_device LANGUAGES CXX)
    find_package(Threads REQUIRED)
endif()

add_executable(xnet_canopen_device main.cpp)
target_compile_features(xnet_canopen_device PRIVATE cxx_std_17)
target_include_directories(xnet_canopen_device PRIVATE ../.. ../../include)
target_link_libraries(xnet_canopen_device PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
if(MSVC)
    target_compile_options(xnet_canopen_device PRIVATE /W4)
else()
    target_compile_options(xnet_canopen_device PRIVATE -Wall -Wextra -Wpedantic)
endif()
